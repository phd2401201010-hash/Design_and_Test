library ieee;
use ieee.std_logic_1164.all;
entity demux_1to4 is
port(DZ4:in std_logic_vector(15 downto 0);
    DS4: in std_logic_vector(1 downto 0);
    P4,Q4,R4,T4:out std_logic_vector(15 downto 0));
end demux_1to4;
architecture bhv of demux_1to4 is
begin
process(DZ4,DS4)is
begin
if(DS4="00")then
T4<=DZ4;
elsif(DS4="01")then
R4<=DZ4;
elsif(DS4="10")then
Q4<=DZ4;
else
P4<=DZ4;
end if;
end process;
end bhv;
