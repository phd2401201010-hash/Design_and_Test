library ieee;
use ieee.std_logic_1164.all;
entity reg is
port(rD:in std_logic_vector(15 downto 0);
		ren: in std_logic;
     rZ: out std_logic_vector(15 downto 0)
	
);
end reg;
architecture bhv of reg is
begin
process(rD,ren)is
begin
if (ren='0')then
rZ<=rD;
end if;
end process;
end bhv;
