library ieee;
use ieee.std_logic_1164.all;
entity demux_1to2 is
port(Z:in std_logic_vector(15 downto 0);
S: in std_logic;
A,B:out std_logic_vector(15 downto 0));
end demux_1to2;
architecture bhv of demux_1to2 is
begin
process(Z,S)is
begin
if(S='0')then
B<=Z;
else
A<=Z;
end if;
end process;
end bhv;
